## 服务器排命令

## M3查询任务

```
docker exec -i vifa-m3-vifa-m3-worker-1 python - <<'PY'
import httpx
from m3.worker.config import Settings
from m3.worker.clients.http import RetryPolicy
from m3.worker.clients.nocobase_api import NocoBaseApiClient

settings = Settings.from_env()
fields = [
    "run_id", "station_id", "status", "interval_seconds",
    "forecast_start", "requested_by", "createdAt", "started_at",
]

with httpx.Client(timeout=15) as client:
    api = NocoBaseApiClient(
        str(settings.nocobase_base_url),
        settings.nocobase_api_key.get_secret_value(),
        client,
        RetryPolicy(max_attempts=1),
    )
    for status in ("queued", "running"):
        rows = api.list_records_all(
            "energy_forecast_manual_runs",
            filter={"status": status},
            fields=fields,
            sort=["-createdAt"],
        )
        print(f"{status}: {len(rows)}")
        for row in rows:
            print({key: row.get(key) for key in fields})
PY
```
